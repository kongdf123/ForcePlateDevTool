2. Update checker
Create src/ForcePlateDevTool/Services/UpdateChecker.cs.
The checker uses the public GitHub Releases API, compares stable versions, and finds the matching installer asset. It does not need a token embedded in the application.

using System.Net.Http;
using System.Net.Http.Headers;
using System.Text.Json;
using System.Text.RegularExpressions;

namespace ForcePlateDevTool.Services;

public sealed record UpdateRelease(
    string TagName,
    string Version,
    string ReleaseUrl,
    string ReleaseNotes,
    string? InstallerUrl);

public sealed class UpdateChecker
{
    private const string LatestReleaseApi =
        "https://api.github.com/repos/kongdf123/ForcePlateDevTool/releases/latest";

    private static readonly HttpClient SharedClient = CreateClient();
    private readonly HttpClient _httpClient;

    public UpdateChecker(HttpClient? httpClient = null)
    {
        _httpClient = httpClient ?? SharedClient;
    }

    private static HttpClient CreateClient()
    {
        var client = new HttpClient
        {
            Timeout = TimeSpan.FromSeconds(5)
        };

        client.DefaultRequestHeaders.UserAgent.Add(
            new ProductInfoHeaderValue("ForcePlateDevTool", "1.0"));

        client.DefaultRequestHeaders.Accept.Add(
            new MediaTypeWithQualityHeaderValue("application/vnd.github+json"));

        return client;
    }

    public async Task<UpdateRelease?> CheckForUpdateAsync(
        string installedVersion,
        CancellationToken cancellationToken = default)
    {
        using var response = await _httpClient.GetAsync(
            LatestReleaseApi, cancellationToken);

        response.EnsureSuccessStatusCode();

        await using var stream =
            await response.Content.ReadAsStreamAsync(cancellationToken);

        using var json = await JsonDocument.ParseAsync(
            stream, cancellationToken: cancellationToken);

        var root = json.RootElement;

        if (root.TryGetProperty("draft", out var draft) &&
            draft.GetBoolean())
        {
            return null;
        }

        if (root.TryGetProperty("prerelease", out var prerelease) &&
            prerelease.GetBoolean())
        {
            return null;
        }

        string tag = root.GetProperty("tag_name").GetString() ?? "";
        if (!TryParseStableVersion(tag, out var latestVersion))
            return null;

        // Release builds may append "+<commit>" to the informational version.
        if (!TryParseStableVersion(installedVersion, out var currentVersion))
            return null;

        if (latestVersion <= currentVersion)
            return null;

        string releaseUrl =
            root.GetProperty("html_url").GetString()
            ?? "https://github.com/kongdf123/ForcePlateDevTool/releases";

        string notes = root.TryGetProperty("body", out var body)
            ? body.GetString() ?? ""
            : "";

        string? installerUrl = null;

        if (root.TryGetProperty("assets", out var assets))
        {
            foreach (var asset in assets.EnumerateArray())
            {
                string name = asset.GetProperty("name").GetString() ?? "";

                if (name.Equals(
                    $"ForcePlateDevTool-Setup-{tag}.exe",
                    StringComparison.OrdinalIgnoreCase))
                {
                    installerUrl =
                        asset.GetProperty("browser_download_url").GetString();
                    break;
                }
            }
        }

        return new UpdateRelease(
            tag,
            latestVersion.ToString(3),
            releaseUrl,
            notes,
            installerUrl);
    }

    public static bool TryParseStableVersion(
        string value,
        out Version version)
    {
        version = new Version(0, 0, 0);

        // Accept v1.2.3 and 1.2.3, with optional build metadata.
        // Reject prereleases such as 1.2.3-rc.1.
        var match = Regex.Match(
            value.Trim(),
            @"^v?(0|[1-9]\d*)\.(0|[1-9]\d*)\.(0|[1-9]\d*)(?:\+[0-9A-Za-z.-]+)?$");

        if (!match.Success)
            return false;

        return Version.TryParse(
            $"{match.Groups[1].Value}.{match.Groups[2].Value}.{match.Groups[3].Value}",
            out version!);
    }
}



Important behavior: if the GitHub API is unreachable, the exception is handled by the UI. The application must continue working normally; update checking must never be a prerequisite for connecting to the device.
3. Persist the “skip this version” preference
Create src/ForcePlateDevTool/Services/UpdatePreferences.cs.
This stores the dismissed version under the current Windows user profile. It does not write into the installation directory, so a non-admin user can use it and upgrades will not overwrite it.

namespace ForcePlateDevTool.Services;

public static class UpdatePreferences
{
    private static readonly string SettingsDirectory = Path.Combine(
        Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
        "ForcePlateDevTool");

    private static readonly string SkippedVersionPath =
        Path.Combine(SettingsDirectory, "SkippedUpdateVersion.txt");

    public static bool IsSkipped(string tag)
    {
        try
        {
            return File.Exists(SkippedVersionPath) &&
                string.Equals(
                    File.ReadAllText(SkippedVersionPath).Trim(),
                    tag,
                    StringComparison.OrdinalIgnoreCase);
        }
        catch (IOException)
        {
            return false;
        }
        catch (UnauthorizedAccessException)
        {
            return false;
        }
    }

    public static void Skip(string tag)
    {
        try
        {
            Directory.CreateDirectory(SettingsDirectory);
            File.WriteAllText(SkippedVersionPath, tag);
        }
        catch (IOException)
        {
            // A preference failure must not affect device diagnostics.
        }
        catch (UnauthorizedAccessException)
        {
            // Keep the application usable even if preferences cannot be saved.
        }
    }
}

using System.Diagnostics;
using System.Windows;
using ForcePlateDevTool.Services;

namespace ForcePlateDevTool.Windows;

public partial class UpdateNotificationWindow : Window
{
    private readonly UpdateRelease _release;

    public UpdateNotificationWindow(UpdateRelease release)
    {
        InitializeComponent();

        _release = release;
        TxtVersion.Text = $"当前版本：{AppVersionInfo.Current.Version}\n" +
                          $"最新版本：{release.Version}";
        TxtReleaseNotes.Text = string.IsNullOrWhiteSpace(release.ReleaseNotes)
            ? "该版本没有提供更新说明。"
            : release.ReleaseNotes;
    }

    private void Download_Click(object sender, RoutedEventArgs e)
    {
        // Open the official installer asset when present; otherwise open
        // the official release page, which also contains the portable ZIP.
        string url = _release.InstallerUrl ?? _release.ReleaseUrl;

        try
        {
            Process.Start(new ProcessStartInfo(url)
            {
                UseShellExecute = true
            });
            Close();
        }
        catch (Exception ex)
        {
            MessageBox.Show(
                $"无法打开下载链接：{ex.Message}\n\n{_release.ReleaseUrl}",
                "打开下载链接失败",
                MessageBoxButton.OK,
                MessageBoxImage.Warning);
        }
    }

    private void RemindLater_Click(object sender, RoutedEventArgs e)
    {
        Close();
    }

    private void SkipVersion_Click(object sender, RoutedEventArgs e)
    {
        UpdatePreferences.Skip(_release.TagName);
        Close();
    }
}



The notification window is non-modal: users can leave it open or dismiss it without blocking the diagnostic interface. The app only opens a browser download; it does not silently install, restart, or close the running application.

private async void BtnCheckUpdates_Click(
    object sender,
    RoutedEventArgs e)
{
    await CheckForUpdatesAsync(manual: true);
}

private async Task CheckForUpdatesAsync(bool manual)
{
    if (_isCheckingForUpdates ||
        (_updateWindow is { IsVisible: true }))
    {
        return;
    }

    _isCheckingForUpdates = true;
    BtnCheckUpdates.IsEnabled = false;

    try
    {
        var currentVersion = AppVersionInfo.Current.Version;

        var release = await _updateChecker.CheckForUpdateAsync(
            currentVersion);

        if (release is null)
        {
            if (manual)
            {
                MessageBox.Show(
                    "当前没有检测到比已安装稳定版本更新的版本。",
                    "检查更新",
                    MessageBoxButton.OK,
                    MessageBoxImage.Information);
            }

            return;
        }

        // Skip-version preference suppresses automatic prompts only.
        // A user-triggered manual check can still show the release.
        if (!manual && UpdatePreferences.IsSkipped(release.TagName))
            return;

        _updateWindow = new UpdateNotificationWindow(release)
        {
            Owner = this
        };

        _updateWindow.Closed += (_, _) => _updateWindow = null;
        _updateWindow.Show();
    }
    catch (OperationCanceledException)
    {
        // An interrupted request should not affect diagnostics.
        if (manual)
        {
            MessageBox.Show(
                "检查更新已取消。",
                "检查更新",
                MessageBoxButton.OK,
                MessageBoxImage.Information);
        }
    }
    catch (HttpRequestException ex)
    {
        if (manual)
        {
            MessageBox.Show(
                $"暂时无法连接 GitHub 检查更新。\n{ex.Message}",
                "检查更新失败",
                MessageBoxButton.OK,
                MessageBoxImage.Warning);
        }
    }
    catch (Exception ex)
    {
        if (manual)
        {
            MessageBox.Show(
                $"检查更新失败：{ex.Message}",
                "检查更新失败",
                MessageBoxButton.OK,
                MessageBoxImage.Warning);
        }
    }
    finally
    {
        _isCheckingForUpdates = false;
        BtnCheckUpdates.IsEnabled = true;
    }
}

      - name: Build installer, ZIP, and SHA-256 checksums
        shell: pwsh
        run: |
          choco install innosetup --no-progress -y

            throw "Failed to install Inno Setup"

          $iscc = "${env:ProgramFiles(x86)}\Inno Setup 6\ISCC.exe"

            throw "Inno Setup compiler not found: $iscc"
          }

          & $iscc "installer/ForcePlateDevTool.iss"
          if ($LASTEXITCODE -ne 0) {
            throw "Installer compilation failed"
          }

          $version = $env:RELEASE_VERSION
          $zip = "artifacts/ForcePlateDevTool-v$version-win-x64.zip"
          $setup = "artifacts/ForcePlateDevTool-Setup-v$version.exe"

          Compress-Archive `
            -Path "artifacts/publish/*" `
            -DestinationPath $zip `
            -CompressionLevel Optimal

          if (-not (Test-Path $setup)) {
            throw "Installer output not found: $setup"
          }

          foreach ($file in @($zip, $setup)) {
            $hash = (Get-FileHash $file -Algorithm SHA256).Hash
            "$hash  $(Split-Path $file -Leaf)" |
              Set-Content "$file.sha256" -Encoding ascii
          }

          "RELEASE_ZIP=$zip" | Out-File $env:GITHUB_ENV -Append
          "RELEASE_SETUP=$setup" | Out-File $env:GITHUB_ENV -Append
          "RELEASE_ZIP_CHECKSUM=$zip.sha256" |
            Out-File $env:GITHUB_ENV -Append
