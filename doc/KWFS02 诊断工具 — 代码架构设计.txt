KWFS02 诊断工具 — 代码架构设计
基于现有生产代码（KW.CommunicationLib / KW.Sensor.DataModel），诊断工具采用分层 + 事件驱动架构，核心原则是：直接复用生产解码器，诊断逻辑与采集逻辑同源。

4. 关键类设计
4.1 SocketLinkerAdapter — 连接层适配
封装生产的 SocketLinker，暴露事件和状态，不修改生产代码：

csharp



// Adapters/SocketLinkerAdapter.cs
public class SocketLinkerAdapter : IAsyncDisposable
{
    private readonly SocketLinker _linker;
    private readonly SocketConfig _config;

    public event EventHandler<byte[]>? BytesReceived;
    public event EventHandler? Disconnected;

    public bool IsConnected => _linker.IsOpen;

    public SocketLinkerAdapter(DiagnosticConfig cfg)
    {
        _config = new SocketConfig
        {
            RemoteIP = cfg.DeviceIp,
            RemotePort = cfg.DevicePort,
            LocalIP = cfg.LocalIp,
            LocalPort = cfg.LocalPort
        };
        _linker = new SocketLinker(_config);
    }

    public async Task<DiagnosticResult> ConnectAsync(CancellationToken ct)
    {
        var sw = Stopwatch.StartNew();
        try
        {
            await Task.Run(() => _linker.Open(), ct);
            sw.Stop();
            return new DiagnosticResult
            {
                LayerName = "Socket连接",
                Status = DiagnosticStatus.Pass,
                Message = $"已连接 {_config.RemoteIP}:{_config.RemotePort}",
                Duration = sw.Elapsed
            };
        }
        catch (Exception ex)
        {
            sw.Stop();
            return new DiagnosticResult
            {
                LayerName = "Socket连接",
                Status = DiagnosticStatus.Fail,
                Message = ex.Message,
                Suggestion = MapExceptionToSuggestion(ex),
                Duration = sw.Elapsed
            };
        }
    }

    public void StartReceiving()
    {
        _linker.OnDataReceive += (bytes) => BytesReceived?.Invoke(this, bytes);
    }

    public int Send(byte[] data) => _linker.Send(data, 0, data.Length);

    private static string MapExceptionToSuggestion(Exception ex) => ex switch
    {
        SocketException { SocketErrorCode: SocketError.TimedOut }
            => "连接超时，请检查设备是否上电、IP/端口是否正确、网线是否插好",
        SocketException { SocketErrorCode: SocketError.ConnectionRefused }
            => "设备拒绝连接，确认设备端服务已启动且端口匹配",
        SocketException { SocketErrorCode: SocketError.NetworkUnreachable }
            => "网络不可达，检查本机网卡 IP 是否与设备同网段",
        _ => "未知错误，请查看日志"
    };
}
4.2 ByteStreamCollector — 原始字节收集 + 统计
csharp



// Engine/ByteStreamCollector.cs
public class ByteStreamCollector
{
    private long _totalBytes;
    private long _totalPackets;
    private long _bytesInWindow;
    private DateTime _windowStart = DateTime.UtcNow;
    private DateTime _lastReceiveTime = DateTime.UtcNow;

    public event EventHandler<byte[]>? BytesCollected;

    public double BytesPerSecond { get; private set; }
    public double PacketsPerSecond { get; private set; }
    public long TotalBytes => _totalBytes;
    public DateTime LastReceiveTime => _lastReceiveTime;

    public void OnBytesReceived(byte[] bytes)
    {
        Interlocked.Add(ref _totalBytes, bytes.Length);
        Interlocked.Increment(ref _totalPackets);
        Interlocked.Add(ref _bytesInWindow, bytes.Length);
        _lastReceiveTime = DateTime.UtcNow;

        if ((_lastReceiveTime - _windowStart).TotalSeconds >= 1)
        {
            BytesPerSecond = Interlocked.Exchange(ref _bytesInWindow, 0);
            PacketsPerSecond = Interlocked.Exchange(ref _totalPackets, 0);
            _windowStart = _lastReceiveTime;
        }

        BytesCollected?.Invoke(this, bytes);
    }

    public bool IsReceiveTimeout(TimeSpan timeout)
        => (DateTime.UtcNow - _lastReceiveTime) > timeout;
}

4.5 DiagnosticPipeline — 顶层编排
csharp



// Engine/DiagnosticPipeline.cs
public class DiagnosticPipeline : IDiagnosticPipeline
{
    private readonly SocketLinkerAdapter _linker;
    private readonly ByteStreamCollector _collector;
    private readonly FrameExtractor _extractor;
    private readonly DecoderRunner _decoder;
    private readonly ZeroOffsetManager _zeroManager;
    private readonly CommandExecutor _commander;

    public DiagnosticPipeline(DiagnosticConfig config)
    {
        Config = config;
        _linker = new SocketLinkerAdapter(config);
        _collector = new ByteStreamCollector();
        _extractor = new FrameExtractor(config);
        _decoder = new DecoderRunner(config);
        _zeroManager = new ZeroOffsetManager(_decoder);
        _commander = new CommandExecutor(_linker);

        // 串联事件链
        _linker.BytesReceived += (_, b) => _collector.OnBytesReceived(b);
        _collector.BytesCollected += (_, b) =>
        {
            RawBytesReceived?.Invoke(this, b);
            _extractor.Feed(b);
        };
        _extractor.FrameExtracted += (_, v) =>
        {
            FrameValidated?.Invoke(this, v);
            if (v.IsValid && v.RawFrame is not null)
            {
                var snapshot = _decoder.Decode(v.RawFrame);
                PhysicalDataDecoded?.Invoke(this, snapshot);
            }
        };
    }

    public async Task<IReadOnlyList<DiagnosticResult>> RunFullDiagnosticAsync(CancellationToken ct)
    {
        var results = new List<DiagnosticResult>();

        // 1. 物理链路
        results.Add(await PhysicalLinkProbe.RunAsync(Config, ct));

        // 2. Socket 连接
        var connResult = await _linker.ConnectAsync(ct);
        results.Add(connResult);
        if (connResult.Status != DiagnosticStatus.Pass) return results;

        // 3. 发送起始命令
        results.Add(await _commander.SendStartCommandAsync(Config.Mode, ct));

        // 4. 等待数据流入 (3 秒)
        results.Add(await WaitForDataAsync(TimeSpan.FromSeconds(3), ct));

        // 5. 帧解析统计
        results.Add(_extractor.GetStatisticsResult());

        // 6. 物理量合理性
        results.Add(CheckPhysicalValueRange());

        foreach (var r in results) LayerCompleted?.Invoke(this, r);
        return results;
    }

    // ...
}

6. ViewModel 层关键设计
6.1 线程安全的数据推送
csharp



// ViewModels/PhysicalDataViewModel.cs
public class PhysicalDataViewModel : ObservableObject
{
    private readonly ConcurrentQueue<PhysicalValueSnapshot> _queue = new();
    private readonly DispatcherTimer _uiTimer;

    public PhysicalValueSnapshot Current { get; private set; }
    public ObservableCollection<PhysicalValueSnapshot> History { get; } = new();

    public PhysicalDataViewModel(IDiagnosticPipeline pipeline)
    {
        pipeline.PhysicalDataDecoded += (_, snapshot) =>
        {
            _queue.Enqueue(snapshot);   // 采集线程：只入队
        };

        _uiTimer = new DispatcherTimer { Interval = TimeSpan.FromMilliseconds(50) };
        _uiTimer.Tick += (_, _) => FlushQueue();
        _uiTimer.Start();
    }

    private void FlushQueue()
    {
        while (_queue.TryDequeue(out var snapshot))
        {
            Current = snapshot;
            History.Add(snapshot);
            if (History.Count > 100) History.RemoveAt(0);
        }
        OnPropertyChanged(nameof(Current));
    }
}
